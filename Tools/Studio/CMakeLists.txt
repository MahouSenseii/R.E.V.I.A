cmake_minimum_required(VERSION 3.20)
project(ReviaClosedPresentationCandidate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.8 COMPONENTS Core REQUIRED)
if(NOT REVIA_CANDIDATE_ROOT OR NOT REVIA_EXPECTED_DIGITS MATCHES "^[123]$")
    message(FATAL_ERROR "A host-approved candidate root and precision are required.")
endif()
add_executable(ReviaStudioDuration "${CMAKE_CURRENT_LIST_DIR}/../../Tests/Fixture/studioDurationTests.cpp")
target_include_directories(ReviaStudioDuration PRIVATE "${REVIA_CANDIDATE_ROOT}/Desktop")
target_compile_definitions(ReviaStudioDuration PRIVATE REVIA_STUDIO_DURATION_STANDALONE)
target_link_libraries(ReviaStudioDuration PRIVATE Qt6::Core)
if(MINGW)
    target_link_options(ReviaStudioDuration PRIVATE -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_test(NAME Revia.ClosedPresentationFormatting COMMAND ReviaStudioDuration ${REVIA_EXPECTED_DIGITS})
set_tests_properties(Revia.ClosedPresentationFormatting PROPERTIES TIMEOUT 20)
